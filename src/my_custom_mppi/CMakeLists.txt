cmake_minimum_required(VERSION 3.16)
project(my_custom_mppi)
find_package(ament_cmake REQUIRED)
find_package(OpenMP REQUIRED)
find_package(ament_index_cpp REQUIRED)
set(deps nav2_core nav2_costmap_2d pluginlib rclcpp rclcpp_lifecycle rcl_interfaces nav_msgs geometry_msgs tf2 tf2_ros tf2_geometry_msgs visualization_msgs)
foreach(dep IN LISTS deps)
  find_package(${dep} REQUIRED)
endforeach()
add_library(custom_mppi_controller SHARED src/controller.cpp src/mppi.cpp)
set_target_properties(custom_mppi_controller PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_compile_features(custom_mppi_controller PRIVATE cxx_std_17)
target_compile_options(custom_mppi_controller PRIVATE -O3 -ffp-contract=off)
target_link_libraries(custom_mppi_controller OpenMP::OpenMP_CXX)
ament_target_dependencies(custom_mppi_controller ${deps} ament_index_cpp)
pluginlib_export_plugin_description_file(nav2_core controller_plugin.xml)
install(TARGETS custom_mppi_controller LIBRARY DESTINATION lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
